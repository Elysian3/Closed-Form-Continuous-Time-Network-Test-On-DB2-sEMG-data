idf_component_register(
    SRCS "../../diagnostics/ads1298_bringup/main/ads1298_bringup.c"
         "../../diagnostics/ads1298_bringup/main/ads1298_protocol.c"
         "../../model/autoncp.c" "model_boot.c"
    INCLUDE_DIRS "../../diagnostics/ads1298_bringup/main" "../../model" "${MODEL_DIR}"
    REQUIRES esp_driver_gpio esp_driver_spi esp_driver_usb_serial_jtag freertos esp_timer esp_mm esp_psram
)
target_compile_definitions(${COMPONENT_LIB} PRIVATE ANTIKYTHERA_MODEL_BOOT_CHECK=1)
option(AUTONCP_MODEL_O3 "Compile only the AutoNCP kernel with -O3" OFF)
if(AUTONCP_MODEL_O3)
    set_source_files_properties("../../model/autoncp.c" PROPERTIES COMPILE_OPTIONS "-O3")
endif()

set(AUTONCP_PROFILE_DIR "" CACHE PATH "Opt-in latency test vectors; empty disables all probes")
option(AUTONCP_STAGE_CLOCKS "Include model stage probes in the benchmark" ON)
option(AUTONCP_INTEGER_FC "Experimental INT8 input / INT32 FC benchmark" OFF)
option(AUTONCP_INTEGER_SIMD "Native S3 vector dot in the integer benchmark" OFF)
option(AUTONCP_BLOCKED_FC "Reuse FP32 inputs across four FC rows" OFF)
if(AUTONCP_BLOCKED_FC)
    if(AUTONCP_INTEGER_FC)
        message(FATAL_ERROR "Blocked FC requires FP32 activations")
    endif()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE AUTONCP_BLOCKED_FC=1)
endif()
if(AUTONCP_INTEGER_FC AND NOT AUTONCP_PROFILE_DIR)
    message(FATAL_ERROR "Integer FC is test-only; provide its reference vectors")
endif()
if(AUTONCP_INTEGER_SIMD AND NOT AUTONCP_INTEGER_FC)
    message(FATAL_ERROR "SIMD FC requires integer FC")
endif()
if(AUTONCP_PROFILE_DIR)
    if(NOT EXISTS "${AUTONCP_PROFILE_DIR}/profile_vectors.h")
        message(FATAL_ERROR "Run profile_autoncp.py prepare first; profile_vectors.h is missing")
    endif()
    target_include_directories(${COMPONENT_LIB} PRIVATE "${AUTONCP_PROFILE_DIR}")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE AUTONCP_BENCHMARK=1)
    if(AUTONCP_INTEGER_FC)
        target_compile_definitions(${COMPONENT_LIB} PRIVATE AUTONCP_INTEGER_FC=1)
        if(AUTONCP_INTEGER_SIMD)
            target_compile_definitions(${COMPONENT_LIB} PRIVATE AUTONCP_INTEGER_SIMD=1)
        endif()
    endif()
    if(AUTONCP_STAGE_CLOCKS)
        target_compile_definitions(${COMPONENT_LIB} PRIVATE AUTONCP_PROFILE=1)
    endif()
endif()
