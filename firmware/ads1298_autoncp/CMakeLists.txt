cmake_minimum_required(VERSION 3.16)
set(MODEL_DIR "" CACHE PATH "Explicit export_autoncp.py output directory")
get_filename_component(MODEL_DIR "${MODEL_DIR}" ABSOLUTE BASE_DIR "${CMAKE_SOURCE_DIR}")
foreach(name autoncp_dims.h autoncp_weights.h autoncp_self_test.h manifest.json)
    if(NOT EXISTS "${MODEL_DIR}/${name}")
        message(FATAL_ERROR "Export the model first and pass -DMODEL_DIR=<directory>; missing ${name}")
    endif()
endforeach()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(ads1298_autoncp)
